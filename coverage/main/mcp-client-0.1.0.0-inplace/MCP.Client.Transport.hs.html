    7 -- | Abstract transport for sending and receiving JSON-RPC messages.
    8 --
    9 -- A record-of-functions rather than a typeclass, keeping 'ClientSession'
   10 -- monomorphic and making it easy to build transports from any I/O backend.
   11 data Transport = Transport
   12   { transport_send :: JSONRPCMessage -> IO ()

   14   , transport_receive :: IO (Maybe JSONRPCMessage)
   15   -- ^ Receive the next message from the server.
   16   -- Returns 'Nothing' on EOF or when the transport is closed.
   17   , transport_close :: IO ()
   18   -- ^ Close the transport and release associated resources.
   19   }
