   41 -- | An active MCP client session over a 'Transport'.
   42 data ClientSession = ClientSession
   43   { session_transport :: Transport
   44   , session_config :: ClientConfig
   45   , session_next_id :: IORef Int
   46   , session_pending :: IORef (IntMap (MVar (Either JSONRPCErrorInfo Value)))
   47   , session_server_caps :: IORef (Maybe ServerCapabilities)
   48   , session_server_info :: IORef (Maybe Implementation)
   49   , session_initialized :: IORef Bool
   50   , session_closed :: IORef Bool
   51   }
   52 
   53 -- | Run an action with a client session.
   54 --
   55 -- Starts a background receive loop that dispatches server responses,
   56 -- requests, and notifications.  On exit the receive thread is killed and
   57 -- all pending requests are failed with 'SessionClosed'.
   58 withClientSession :: Transport -> ClientConfig -> (ClientSession -> IO a) -> IO a
   59 withClientSession transport cfg action = do
   60   session <- newSession transport cfg
   61   bracket
   62     (forkIO $ receiveLoop session)
   63     ( \tid -> do
   64         killThread tid
   65         writeIORef (session_closed session) True
   66         failAllPending session
   67     )
   68     (const $ action session)

   70 -- | Perform the MCP initialization handshake.
   71 --
   72 -- Sends @initialize@, validates the server's protocol version, stores
   73 -- the server capabilities, and sends @notifications\/initialized@.
   74 initialize :: ClientSession -> IO InitializeResult
   75 initialize session = do
   76   let cfg = session_config session
   77       params =
   78         InitializeParams
   79           { protocolVersion = pROTOCOL_VERSION
   80           , capabilities = config_capabilities cfg
   81           , clientInfo = config_client_info cfg
   82           }
   83   result <- sendRequest session "initialize" (toJSON params)
   84   case Aeson.fromJSON result of
   85     Aeson.Error err -> throwIO $ DecodeError (T.pack err)
   86     Aeson.Success (ir :: InitializeResult) -> do
   87       let InitializeResult{protocolVersion = sv} = ir
   88       if sv /= pROTOCOL_VERSION
   89         then throwIO $ ProtocolVersionMismatch pROTOCOL_VERSION sv
   90         else do
   91           let InitializeResult
   92                 { capabilities = caps
   93                 , serverInfo = info
   94                 } = ir
   95           writeIORef (session_server_caps session) (Just caps)
   96           writeIORef (session_server_info session) (Just info)
   97           writeIORef (session_initialized session) True
   98           -- Send initialized notification
   99           sendNotification session $
  100             NotificationMessage $
  101               JSONRPCNotification rPC_VERSION "notifications/initialized" Aeson.Null
  102           return ir
  103

  114 newSession :: Transport -> ClientConfig -> IO ClientSession
  115 newSession transport cfg = do
  116   next_id <- newIORef 1
  117   pending <- newIORef IM.empty
  118   caps <- newIORef Nothing
  119   info <- newIORef Nothing
  120   inited <- newIORef False
  121   closed <- newIORef False
  122   return
  123     ClientSession
  124       { session_transport = transport
  125       , session_config = cfg
  126       , session_next_id = next_id
  127       , session_pending = pending
  128       , session_server_caps = caps
  129       , session_server_info = info
  130       , session_initialized = inited
  131       , session_closed = closed
  132       }

  134 -- | Send a JSON-RPC request and block until the response arrives.
  135 sendRequest :: ClientSession -> Text -> Value -> IO Value
  136 sendRequest session method params = do
  137   closed <- readIORef (session_closed session)
  138   if closed
  139     then throwIO SessionClosed
  140     else do
  141       (req_id_int, mvar) <- registerPending session
  142       let req_id = RequestId (toJSON req_id_int)
  143           msg =
  144             RequestMessage $
  145               JSONRPCRequest rPC_VERSION req_id method params
  146       transport_send (session_transport session) msg
  147       let timeout_us = config_request_timeout_us (session_config session)
  148       result <-
  149         if timeout_us <= 0
  150           then Just <$> takeMVar mvar
  151           else timeout timeout_us (takeMVar mvar)
  152       -- Clean up pending entry
  153       atomicModifyIORef' (session_pending session) $ \m ->
  154         (IM.delete req_id_int m, ())
  155       case result of
  156         Nothing -> throwIO $ RequestTimeout req_id
  157         Just (Left (JSONRPCErrorInfo err_code err_msg err_data)) ->
  158           throwIO $ RPCError err_code err_msg err_data
  159         Just (Right val) -> return val
  160 
  161 -- | Send a notification (fire-and-forget, no response expected).
  162 sendNotification :: ClientSession -> JSONRPCMessage -> IO ()
  163 sendNotification session msg =
  164   transport_send (session_transport session) msg

  166 -- | Allocate a fresh request ID and MVar for the response.
  167 registerPending :: ClientSession -> IO (Int, MVar (Either JSONRPCErrorInfo Value))
  168 registerPending session = do
  169   mvar <- newEmptyMVar
  170   req_id <- atomicModifyIORef' (session_next_id session) $ \n -> (n + 1, n)
  171   atomicModifyIORef' (session_pending session) $ \m ->
  172     (IM.insert req_id mvar m, ())
  173   return (req_id, mvar)
  174 
  175 -- | Fill all pending MVars with a 'SessionClosed' error so no thread hangs.
  176 failAllPending :: ClientSession -> IO ()
  177 failAllPending session = do
  178   pending <- atomicModifyIORef' (session_pending session) $ \m -> (IM.empty, m)
  179   let err = JSONRPCErrorInfo iNTERNAL_ERROR "Session closed" Nothing
  180   mapM_ (\mvar -> putMVar mvar (Left err)) pending
  181 
  182 -- | Background loop reading from the transport and dispatching messages.
  183 receiveLoop :: ClientSession -> IO ()
  184 receiveLoop session = do
  185   mb_msg <- try $ transport_receive (session_transport session)
  186   case mb_msg of
  187     Left (_ :: SomeException) -> return ()
  188     Right Nothing -> return ()
  189     Right (Just msg) -> do
  190       dispatchMessage session msg
  191       receiveLoop session
  192 
  193 -- | Route an incoming message to the appropriate handler.
  194 dispatchMessage :: ClientSession -> JSONRPCMessage -> IO ()
  195 dispatchMessage session = \case
  196   ResponseMessage (JSONRPCResponse _ req_id result_val) ->
  197     deliverResponse session req_id (Right result_val)
  198   ErrorMessage (JSONRPCError _ req_id err_info) ->
  199     deliverResponse session req_id (Left err_info)
  200   RequestMessage (JSONRPCRequest _ req_id method params) ->
  201     handleServerRequest session req_id method params
  202   NotificationMessage (JSONRPCNotification _ method params) ->
  203     handleServerNotification session method params
  204 
  205 -- | Deliver a response to the waiting caller.
  206 deliverResponse :: ClientSession -> RequestId -> Either JSONRPCErrorInfo Value -> IO ()
  207 deliverResponse session (RequestId req_id_val) result = do
  208   let mb_int = case req_id_val of
  209         Aeson.Number n -> Just (round n :: Int)
  210         _ -> Nothing
  211   case mb_int of
  212     Nothing -> return ()
  213     Just req_id_int -> do
  214       mb_mvar <- atomicModifyIORef' (session_pending session) $ \m ->
  215         case IM.lookup req_id_int m of
  216           Nothing -> (m, Nothing)
  217           Just mvar -> (IM.delete req_id_int m, Just mvar)
  218       case mb_mvar of
  219         Nothing -> return ()
  220         Just mvar -> putMVar mvar result
  221 
  222 -- | Handle a request from the server (ping, sampling, roots, elicitation).
  223 handleServerRequest :: ClientSession -> RequestId -> Text -> Value -> IO ()
  224 handleServerRequest session req_id method params = do
  225   let cfg = session_config session
  226       send_response val =
  227         sendNotificationRaw session $
  228           ResponseMessage $ JSONRPCResponse rPC_VERSION req_id val
  229       send_error err_code err_msg =
  230         sendNotificationRaw session $
  231           ErrorMessage $
  232             JSONRPCError rPC_VERSION req_id (JSONRPCErrorInfo err_code err_msg Nothing)
  233       dispatch :: (FromJSON a, ToJSON b) => Maybe (a -> IO b) -> IO ()
  234       dispatch Nothing = send_error mETHOD_NOT_FOUND ("Method not found: " <> method)
  235       dispatch (Just handler) =
  236         case Aeson.fromJSON params of
  237           Aeson.Error err -> send_error iNVALID_PARAMS (T.pack err)
  238           Aeson.Success p -> do
  239             result <- handler p
  240             send_response (toJSON result)
  241   case method of
  242     "ping" -> send_response (Aeson.Object mempty)
  243     "sampling/createMessage" -> dispatch (config_on_create_message cfg)
  244     "roots/list" -> dispatch (config_on_list_roots cfg)
  245     "elicitation/create" -> dispatch (config_on_elicit cfg)
  246     _ -> send_error mETHOD_NOT_FOUND ("Method not found: " <> method)
  247 
  248 -- | Handle a notification from the server.
  249 handleServerNotification :: ClientSession -> Text -> Value -> IO ()
  250 handleServerNotification session method params = do
  251   let cfg = session_config session
  252       tryCall :: (FromJSON a) => Maybe (a -> IO ()) -> IO ()
  253       tryCall Nothing = callFallback
  254       tryCall (Just handler) =
  255         case Aeson.fromJSON params of
  256           Aeson.Error _ -> callFallback
  257           Aeson.Success p -> handler p
  258       callFallback =
  259         case config_on_fallback cfg of
  260           Nothing -> return ()
  261           Just fb -> fb $ NotificationMessage $ JSONRPCNotification rPC_VERSION method params
  262   case method of
  263     "notifications/progress" -> tryCall (config_on_progress cfg)
  264     "notifications/message" -> tryCall (config_on_logging_message cfg)
  265     "notifications/resources/updated" -> tryCall (config_on_resource_updated cfg)
  266     "notifications/resources/list_changed" ->
  267       maybe callFallback id (config_on_resource_list_changed cfg)
  268     "notifications/tools/list_changed" ->
  269       maybe callFallback id (config_on_tool_list_changed cfg)
  270     "notifications/prompts/list_changed" ->
  271       maybe callFallback id (config_on_prompt_list_changed cfg)
  272     "notifications/cancelled" -> tryCall (config_on_cancelled cfg)
  273     _ -> callFallback
  274 
  275 -- | Raw send — used internally for responses to server requests.
  276 sendNotificationRaw :: ClientSession -> JSONRPCMessage -> IO ()
  277 sendNotificationRaw session msg =
  278   transport_send (session_transport session) msg
