    9 -- | Configuration for a client session.
   10 data ClientConfig = ClientConfig
   11   { config_client_info :: Implementation
   12   -- ^ Client name and version sent during initialization.
   13   , config_capabilities :: ClientCapabilities
   14   -- ^ Client capabilities to advertise to the server.
   15   , config_request_timeout_us :: Int
   16   -- ^ Default per-request timeout in microseconds (0 = no timeout).
   17   , config_on_create_message :: Maybe (CreateMessageParams -> IO CreateMessageResult)
   18   -- ^ Handler for server-initiated @sampling\/createMessage@ requests.
   19   , config_on_list_roots :: Maybe (ListRootsParams -> IO ListRootsResult)
   20   -- ^ Handler for server-initiated @roots\/list@ requests.
   21   , config_on_elicit :: Maybe (ElicitParams -> IO ElicitResult)
   22   -- ^ Handler for server-initiated @elicitation\/create@ requests.
   23   , config_on_progress :: Maybe (ProgressParams -> IO ())
   24   -- ^ Handler for @notifications\/progress@ from the server.
   25   , config_on_logging_message :: Maybe (LoggingMessageParams -> IO ())
   26   -- ^ Handler for @notifications\/message@ (logging) from the server.
   27   , config_on_resource_updated :: Maybe (ResourceUpdatedParams -> IO ())
   28   -- ^ Handler for @notifications\/resources\/updated@ from the server.
   29   , config_on_resource_list_changed :: Maybe (IO ())
   30   -- ^ Handler for @notifications\/resources\/list_changed@ from the server.
   31   , config_on_tool_list_changed :: Maybe (IO ())
   32   -- ^ Handler for @notifications\/tools\/list_changed@ from the server.
   33   , config_on_prompt_list_changed :: Maybe (IO ())
   34   -- ^ Handler for @notifications\/prompts\/list_changed@ from the server.
   35   , config_on_cancelled :: Maybe (CancelledParams -> IO ())
   36   -- ^ Handler for @notifications\/cancelled@ from the server.
   37   , config_on_fallback :: Maybe (JSONRPCMessage -> IO ())
   38   -- ^ Catch-all handler for unrecognized messages.
   39   }
   40
