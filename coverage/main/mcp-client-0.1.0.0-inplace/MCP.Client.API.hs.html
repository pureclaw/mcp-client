   63 -- | Send a @ping@ request.
   64 ping :: ClientSession -> IO ()
   65 ping session = do
   66   (_ :: Value) <- sendRequestTyped session "ping" (toJSON (PingParams Nothing))
   67   return ()

   69 -- | List available tools.  Pass 'Nothing' for the first page.
   70 listTools :: ClientSession -> Maybe Cursor -> IO ListToolsResult
   71 listTools session cursor =
   72   sendRequestTyped session "tools/list" (toJSON (ListToolsParams cursor))

   74 -- | Call a tool by name with optional arguments.

  125 -- * Client-to-server notifications

  127 -- | Notify the server that a request has been cancelled.
  128 sendCancelled :: ClientSession -> RequestId -> Maybe Text -> IO ()
  129 sendCancelled session req_id reason =
  130   Session.sendNotification session $
  131     NotificationMessage $
  132       JSONRPCNotification rPC_VERSION "notifications/cancelled" $
  133         toJSON (CancelledParams req_id reason)
  134 
  135 -- | Send a progress notification to the server.
  136 sendProgress :: ClientSession -> ProgressParams -> IO ()
  137 sendProgress session params =
  138   Session.sendNotification session $
  139     NotificationMessage $
  140       JSONRPCNotification rPC_VERSION "notifications/progress" $
  141         toJSON params
  142 
  143 -- | Notify the server that the client's root list has changed.
  144 sendRootsListChanged :: ClientSession -> IO ()
  145 sendRootsListChanged session =

  147     NotificationMessage $
  148       JSONRPCNotification rPC_VERSION "notifications/roots/list_changed" Aeson.Null

  150 -- * Internal

  152 sendRequestTyped :: (FromJSON a) => ClientSession -> Text -> Value -> IO a
  153 sendRequestTyped session method params = do
  154   val <- Session.sendRequest session method params
  155   case Aeson.fromJSON val of
  156     Aeson.Error err -> throwIO $ DecodeError (T.pack err)
  157     Aeson.Success a -> return a
